cmake_minimum_required(VERSION 3.22)
option(STM_SD_WITH_SDMMC "Build STM32 HAL SDMMC adapter" OFF)
if(STM_SD_WITH_SDMMC)
    add_library(stm_sd_sdmmc STATIC sdmmc.c)
    target_include_directories(stm_sd_sdmmc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(stm_sd_sdmmc PUBLIC stm_sd)
    if(TARGET stm32cubemx)
        target_link_libraries(stm_sd_sdmmc PUBLIC stm32cubemx)
    endif()
endif()
